        function initStickyWingAd() {
            try {
                var rightSpaceBanner = document.querySelector('#spaceBanner .rightSpaceBanner');
                if (!rightSpaceBanner) return;
    
                var sponsorBar = document.getElementById('ad_sponsorBar');
                var offsetTop = sponsorBar ? sponsorBar.getBoundingClientRect().top + window.pageYOffset : 0;
                var isFixed = false;
                var fixedLeft = 0;

                function updateLeft() {
                    fixedLeft = Math.floor((document.documentElement.clientWidth + 970) / 2) + 17;
                    if (isFixed) rightSpaceBanner.style.left = fixedLeft + 'px';
                }

                function onScroll() {
                    var scrollY = window.pageYOffset || document.documentElement.scrollTop;
                    if (scrollY >= offsetTop - 16) {
                        if (!isFixed) {
                            updateLeft();
                            rightSpaceBanner.style.position = 'fixed';
                            rightSpaceBanner.style.top = '16px';
                            rightSpaceBanner.style.left = fixedLeft + 'px';
                            rightSpaceBanner.style.right = 'auto';
                            rightSpaceBanner.style.marginLeft = '0';
                            isFixed = true;
                        }
                    } else if (isFixed) {
                        rightSpaceBanner.style.position = '';
                        rightSpaceBanner.style.top = '';
                        rightSpaceBanner.style.left = '';
                        rightSpaceBanner.style.right = '';
                        rightSpaceBanner.style.marginLeft = '';
                        isFixed = false;
                    }
                }

                if (window.addEventListener) {
                    window.addEventListener('scroll', onScroll, false);
                    window.addEventListener('resize', updateLeft, false);
                } else {
                    window.attachEvent('onscroll', onScroll);
                    window.attachEvent('onresize', updateLeft);
                }
            } catch(e) {}
        }

        function checkSubmit() {
            try {
                var nIsNow = $("#ps").val();
                if (nIsNow != '3') {
                    return true;
                }

                var ps1 = $("#search-option form input[name='ps1']").attr('value');
				var ps2 = $("#search-option form input[name='ps2']").attr('value');
                bValidInput = true;
                if (ps1.length < 8 || ps2.length < 8) {
                    bValidInput = false;
                }
                for (i = 0; i < ps1.length; i++) {
                    cChar = ps1.charCodeAt(i);
                    if (cChar < 48 || cChar > 57) {
                        bValidInput = false;
                        break;
                    }
                }
                for (i = 0; i < ps2.length; i++) {
                    cChar = ps2.charCodeAt(i);
                    if (cChar < 48 || cChar > 57) {
                        bValidInput = false;
                        break;
                    }
                }

                if (ps1 > ps2) {
                    bValidInput = false;
                }

                //날짜유효성 체크
                var bCheckps1 = isDate(ps1);
                var bCheckps2 = isDate(ps2);
				if (!(bCheckps1 == true && bCheckps2 == true)) {
					bValidInput = false;
				}
				
                if (bValidInput == false) {
                    alert('검색 기간이 올바르지 않습니다');
                   // $("#search-option form input[name='ps1']").focus();
                    return false;
                }
                return true;
            } catch (e) {
                return true;
            }
        }

        function setRefreshTitle(params) {
            if (refreshed == 1) {
                return;
            }
            var refreshPeriod;
            try {
                document.forms.refresh_option.refresh_option1;
            } catch (e) {
                return;
            }

                document.forms.refresh_option.refresh_option2.value;
            var remainSecond = refreshPeriod - timeCount;
            if (refreshPeriod > 0 && remainSecond <= 0) {
                document.forms.refresh_option.refresh.value = refreshPeriod;
                var refreshPeriodQuery = "r=" + refreshPeriod;
                var refreshUrl = "?" + params;
                if (refreshUrl.indexOf(refreshPeriodQuery) < 0) {
                    refreshUrl += "&" + refreshPeriodQuery;
                }
                self.location.href = refreshUrl;
                refreshed = 1;
            }
            if (refreshPeriod > 0) {
                remainMinute = Math.floor(remainSecond / 60);
                remainMinute = remainMinute < 10 ? "0" + remainMinute : remainMinute;
                remainSecond = Math.floor(remainSecond % 60);
